   **Example Code Snippet** (Add this to main.rs or import as a module):
   ```
   use std::io::Cursor;
   use tauri::{command, State};
   use winreg::enums::*;
   use winreg::RegKey;
   use exeico::{extract_icon, Icon};  // Assuming exeico crate; adjust for windows-icons
   use image::io::Reader as ImageReader;
   use base64::{encode};

   #[derive(serde::Serialize)]
   struct AppInfo {
       name: String,
       path: Option<String>,
       icon_base64: Option<String>,
   }

   #[command]
   pub fn get_installed_apps() -> Result<Vec<AppInfo>, String> {
       #[cfg(not(target_os = "windows"))]
       return Err("This function is Windows-only".to_string());

       #[cfg(target_os = "windows")]
       {
           let hklm = RegKey::predef(HKEY_LOCAL_MACHINE);
           let keys = vec![
               r"SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
               r"SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall",
           ];

           let mut apps: Vec<AppInfo> = Vec::new();

           for key_path in keys {
               if let Ok(uninstall) = hklm.open_subkey(key_path) {
                   for subkey_name in uninstall.enum_keys().map(|x| x.unwrap()) {
                       if let Ok(subkey) = uninstall.open_subkey(&subkey_name) {
                           let name: Option<String> = subkey.get_value("DisplayName").ok();
                           let path: Option<String> = subkey.get_value("InstallLocation").ok().or_else(|| subkey.get_value("DisplayIcon").ok());
                           let display_icon: Option<String> = subkey.get_value("DisplayIcon").ok();

                           if let Some(name) = name {
                               let mut icon_base64 = None;
                               if let Some(icon_str) = display_icon {
                                   let parts: Vec<&str> = icon_str.split(',').collect();
                                   let exe_path = parts[0].trim_matches('"');
                                   let index: usize = parts.get(1).and_then(|s| s.parse().ok()).unwrap_or(0);

                                   // Extract icon
                                   if let Ok(icons) = extract_icon(exe_path) {
                                       if let Some(icon) = icons.get(index) {
                                           // Convert to PNG and base64
                                           let mut buffer = Vec::new();
                                           icon.write_png(&mut buffer).map_err(|e| e.to_string())?;
                                           icon_base64 = Some(encode(&buffer));
                                       }
                                   }
                               }

                               apps.push(AppInfo { name, path, icon_base64 });
                           }
                       }
                   }
               }
           }

           Ok(apps)
       }
   }
   ```
